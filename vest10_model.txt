//@version=6
// Vest $10 model — BTC 1-hour pullback WITH the daily trend (shorts while the daily trend is down; longs off until tested)
// Rules (tested Nov 2025 – Nov 2026 with Vest costs: 0.01% taker + 0.01% slippage per fill):
//   Daily trend = prior daily close vs its 20-day EMA. Only when DOWN:
//   previous 1h candle closed UP (against the trend), this 1h candle closes DOWN → SHORT at this close.
//   Stop = highest high of the two candles (skip if the stop is closer than 0.4% of price). Target = 2R. Time exit after 48 hours.
// Entry filter: the down candle must be solid — body more than half its high-low range.
// Exit: 2R target; once +0.5R, stop moves to just past entry. Tested (one trade at a time): ~0.9 trades/day, 70% won, ≈ +0.07R/trade after costs.
// 1-hour ONLY: the same rules on 15-minute candles, or a 1h candle re-checked every 15 min, lost money in testing.
// Chart: BINANCE:BTCUSDT, 1-hour. Simulation only — no real orders.
// Tip: Settings → Style → untick "Trades on chart" (this script draws its own clean entry / stop / target boxes).
strategy("Vest $10 model (BTC 1h)", shorttitle = "Vest10", overlay = true, process_orders_on_close = true, pyramiding = 0,
     initial_capital = 500, default_qty_type = strategy.fixed, default_qty_value = 0.01,
     commission_type = strategy.commission.percent, commission_value = 0.02,
     max_boxes_count = 200, max_labels_count = 300, max_lines_count = 300)
riskUsd   = input.float(2.0, "Risk per trade ($)", minval = 0.5, step = 0.5)
allowLong = input.bool(false, "Also take longs when the daily trend is UP (untested — off)")
minStop   = input.float(0.4, "Minimum stop distance (% of price)", step = 0.1)
rr        = input.float(2.0, "Target (R)", step = 0.5)
needBody  = input.bool(true, "Only solid signal candles (body > half the range)", tooltip = "Tested with +0.5R: 69% won, +0.052R/trade (halves +0.021 / +0.092). Without it: 69% won, +0.029R.")
beAt      = input.float(0.5, "Move stop to entry once the trade reaches +x R (0 = off)", minval = 0, step = 0.5,
     tooltip = "With the solid-candle filter: +0.5R → ~70% won; +1R → ~55% won but more per trade. Off → ~38% won.")
showPanel = input.bool(true, "Show info panel (top right)")

fp(float x) => str.tostring(x, "#,###")
fq(float x) => str.tostring(x, "#.####")

int dTrend = request.security(syminfo.tickerid, "D", close[1] < ta.ema(close, 20)[1] ? -1 : 1, lookahead = barmerge.lookahead_on)
bool shortSig = dTrend == -1 and close[1] > open[1] and close < open and (not needBody or (open - close) > 0.5 * (high - low))
bool longSig  = allowLong and dTrend == 1 and close[1] < open[1] and close > open and (not needBody or (close - open) > 0.5 * (high - low))
float stS = math.max(high, high[1])
float stL = math.min(low, low[1])
bool flat = strategy.position_size == 0

var int   entryBar = na
var float eEntry = na
var float eRisk = na
var float eStop = na
var float eTgt = na
var float eQty = na
var int   eDir = 0
var bool  beDone = false
var box   bStop = na
var box   bTgt = na
var line  lEntry = na
var line  lBE = na
var int   nClosed = 0

// 1. a trade just closed → result label, freeze its boxes
if strategy.closedtrades > nClosed
    int k = strategy.closedtrades - 1
    float rRes = strategy.closedtrades.profit(k) / riskUsd
    int xb = strategy.closedtrades.exit_bar_index(k)
    color rc = rRes > 0.05 ? color.new(color.green, 0) : rRes < -0.05 ? color.new(color.red, 0) : color.new(color.gray, 0)
    label.new(xb, strategy.closedtrades.exit_price(k), (rRes >= 0 ? "+" : "") + str.tostring(rRes, "0.0") + "R",
         style = label.style_label_left, color = rc, textcolor = color.white, size = size.small)
    box.set_right(bStop, xb)
    box.set_right(bTgt, xb)
    line.set_x2(lEntry, xb)
    line.set_x2(lBE, xb)
    nClosed := strategy.closedtrades
    eDir := 0

// 2. new signal
int sigDir = 0
float sigStop = na
if flat and shortSig and (stS - close) / close * 100 >= minStop
    sigDir := -1
    sigStop := stS
else if flat and longSig and (close - stL) / close * 100 >= minStop
    sigDir := 1
    sigStop := stL

if sigDir != 0
    float risk = math.abs(close - sigStop)
    float tgt = close + sigDir * rr * risk
    float q = riskUsd / risk
    if sigDir == -1
        strategy.entry("S", strategy.short, qty = q, comment = "SELL")
        strategy.exit("SX", "S", stop = sigStop, limit = tgt, comment_profit = "TP", comment_loss = "SL")
    else
        strategy.entry("L", strategy.long, qty = q, comment = "BUY")
        strategy.exit("LX", "L", stop = sigStop, limit = tgt, comment_profit = "TP", comment_loss = "SL")
    entryBar := bar_index
    eEntry := close
    eRisk := risk
    eStop := sigStop
    eTgt := tgt
    eQty := q
    eDir := sigDir
    beDone := false
    lBE := na
    bStop := box.new(bar_index, math.max(close, sigStop), bar_index + 1, math.min(close, sigStop), border_color = color.new(color.red, 50), bgcolor = color.new(color.red, 80))
    bTgt := box.new(bar_index, math.max(close, tgt), bar_index + 1, math.min(close, tgt), border_color = color.new(color.green, 50), bgcolor = color.new(color.green, 80))
    lEntry := line.new(bar_index, close, bar_index + 1, close, color = color.white, width = 2)
    string txt = (sigDir == -1 ? "SELL  " : "BUY  ") + fp(close) + "\nStop  " + fp(sigStop) + "\nTarget  " + fp(tgt) + "\nSize  " + fq(q) + " BTC"
    label.new(bar_index, sigStop, txt, style = sigDir == -1 ? label.style_label_down : label.style_label_up,
         color = sigDir == -1 ? color.new(color.red, 0) : color.new(color.green, 0), textcolor = color.white, size = size.normal, textalign = text.align_left)
    alert("Vest10 " + (sigDir == -1 ? "SELL" : "BUY") + " BTC @ " + fp(close) + " | stop " + fp(sigStop) + " | target " + fp(tgt) +
         " | size " + fq(q) + " BTC ($" + str.tostring(riskUsd) + " risk) | move stop at " + fp(close + sigDir * beAt * risk), alert.freq_once_per_bar_close)

// 3. manage the open trade
if not flat and eDir != 0 and bar_index > entryBar
    box.set_right(bStop, bar_index + 1)
    box.set_right(bTgt, bar_index + 1)
    line.set_x2(lEntry, bar_index + 1)
    line.set_x2(lBE, bar_index + 1)
    bool reached = eDir == -1 ? low <= eEntry - beAt * eRisk : high >= eEntry + beAt * eRisk
    if beAt > 0 and not beDone and reached
        float ns = eEntry + eDir * 0.1 * eRisk
        if eDir == -1
            strategy.exit("SX", "S", stop = ns, limit = eTgt, comment_profit = "TP", comment_loss = "BE")
        else
            strategy.exit("LX", "L", stop = ns, limit = eTgt, comment_profit = "TP", comment_loss = "BE")
        beDone := true
        eStop := ns
        lBE := line.new(bar_index, ns, bar_index + 1, ns, color = color.orange, width = 2, style = line.style_dashed)
        label.new(bar_index, ns, "Move stop → " + fp(ns), style = eDir == -1 ? label.style_label_up : label.style_label_down,
             color = color.new(color.orange, 0), textcolor = color.black, size = size.small)
        alert("Vest10: MOVE STOP to " + fp(ns) + " (trade reached +" + str.tostring(beAt) + "R)", alert.freq_once_per_bar_close)
    if bar_index - entryBar >= 48
        strategy.close_all(comment = "48h")
        alert("Vest10: CLOSE the trade now (48 hours passed)", alert.freq_once_per_bar_close)

// 4. info panel
var table T = table.new(position.top_right, 2, 7, bgcolor = color.new(#131722, 5), frame_color = color.new(color.gray, 60), frame_width = 1)
if showPanel and barstate.islast
    bool inT = not flat and eDir != 0
    bool wrongTf = timeframe.period != "60"
    string st = wrongTf ? "Switch chart to 1h" : inT ? (eDir == -1 ? "IN SHORT" : "IN LONG") : dTrend == -1 ? "Waiting for signal" : "Off: daily trend up"
    color sc = wrongTf ? color.red : inT ? color.orange : dTrend == -1 ? color.silver : color.gray
    float openR = inT ? (close - eEntry) * eDir / eRisk : na
    table.cell(T, 0, 0, "Vest $10 model", text_color = color.orange, text_halign = text.align_left, text_size = size.normal)
    table.cell(T, 1, 0, st, text_color = sc, text_halign = text.align_right, text_size = size.normal)
    table.cell(T, 0, 1, "Entry", text_color = color.gray, text_halign = text.align_left)
    table.cell(T, 1, 1, inT ? fp(eEntry) : "—", text_color = color.white, text_halign = text.align_right)
    table.cell(T, 0, 2, "Stop", text_color = color.gray, text_halign = text.align_left)
    table.cell(T, 1, 2, inT ? fp(eStop) + (beDone ? " (moved)" : "") : "—", text_color = color.red, text_halign = text.align_right)
    table.cell(T, 0, 3, "Target", text_color = color.gray, text_halign = text.align_left)
    table.cell(T, 1, 3, inT ? fp(eTgt) : "—", text_color = color.green, text_halign = text.align_right)
    table.cell(T, 0, 4, "Move stop when price hits", text_color = color.gray, text_halign = text.align_left)
    table.cell(T, 1, 4, inT and not beDone and beAt > 0 ? fp(eEntry + eDir * beAt * eRisk) : "—", text_color = color.orange, text_halign = text.align_right)
    table.cell(T, 0, 5, "Size", text_color = color.gray, text_halign = text.align_left)
    table.cell(T, 1, 5, inT ? fq(eQty) + " BTC" : "—", text_color = color.white, text_halign = text.align_right)
    table.cell(T, 0, 6, "Open P/L", text_color = color.gray, text_halign = text.align_left)
    table.cell(T, 1, 6, inT ? (openR >= 0 ? "+" : "") + str.tostring(openR, "0.00") + "R" : "—", text_color = inT and openR >= 0 ? color.green : color.red, text_halign = text.align_right)

bgcolor(dTrend == -1 ? color.new(color.red, 96) : na, title = "Daily trend down")
