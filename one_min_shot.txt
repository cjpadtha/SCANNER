//@version=6
// 1-minute shot (Vest instant $10 account) — a LOTTERY trade, not an edge.
// Tested (6 months of BTC 1m, Vest costs): target 0.45% / stop 0.15% → ~1 in 4 hit the target, usually in 10–15 min; avg −0.10R per try.
// Why do it at all: on the $10 instant account the loss is capped at the fee and payouts are on demand, so a win (+~$21) can be withdrawn and the account resets.
// Use: BTC 1-minute chart. Inside the window (default US open 8:30–10:00 ET) the panel shows side, size, stop and target for the CURRENT price.
// Enter at market, set the SL and TP on Vest at exactly these prices, then leave it alone. One shot at a time.
indicator("1-minute shot (Vest $10)", shorttitle = "1m shot", overlay = true, max_lines_count = 10, max_labels_count = 10)
riskUsd = input.float(9.0, "Max loss incl. fees ($) — keep under the $10 drawdown", step = 0.5)
stopPct = input.float(0.15, "Stop distance (%)", step = 0.05)
tgtPct  = input.float(0.45, "Target distance (%)", step = 0.05)
sideIn  = input.string("With daily trend", "Side", options = ["With daily trend", "Short", "Long"])
feePct  = input.float(0.03, "Fees + slippage, round trip (%)", step = 0.01)
sess    = input.session("0830-1000", "Window (exchange-independent, New York time)")

fp(float x) => str.tostring(x, "#,###.0")
int dTrend = request.security(syminfo.tickerid, "D", close[1] < ta.ema(close, 20)[1] ? -1 : 1, lookahead = barmerge.lookahead_on)
int side = sideIn == "Short" ? -1 : sideIn == "Long" ? 1 : dTrend
bool inWin = not na(time(timeframe.period, sess, "America/New_York"))
float sl = close * (1 - side * stopPct / 100)
float tp = close * (1 + side * tgtPct / 100)
float qty = riskUsd / (close * (stopPct + feePct) / 100)
float win = qty * close * (tgtPct - feePct) / 100

bgcolor(inWin ? color.new(color.yellow, 92) : na, title = "Shot window")
var line lSL = na
var line lTP = na
var table T = table.new(position.top_right, 2, 6, bgcolor = color.new(#131722, 5), frame_color = color.new(color.gray, 60), frame_width = 1)
if barstate.islast
    line.delete(lSL)
    line.delete(lTP)
    lSL := line.new(bar_index - 5, sl, bar_index + 10, sl, color = color.red, width = 2)
    lTP := line.new(bar_index - 5, tp, bar_index + 10, tp, color = color.green, width = 2)
    table.cell(T, 0, 0, "1-minute shot", text_color = color.orange, text_halign = text.align_left)
    table.cell(T, 1, 0, inWin ? (side == -1 ? "SHORT now" : "LONG now") : "Outside window", text_color = inWin ? (side == -1 ? color.red : color.green) : color.gray, text_halign = text.align_right)
    table.cell(T, 0, 1, "Size", text_color = color.gray, text_halign = text.align_left)
    table.cell(T, 1, 1, str.tostring(qty, "#.###") + " BTC", text_color = color.white, text_halign = text.align_right)
    table.cell(T, 0, 2, "Stop loss", text_color = color.gray, text_halign = text.align_left)
    table.cell(T, 1, 2, fp(sl) + "  (−$" + str.tostring(riskUsd, "#.0") + ")", text_color = color.red, text_halign = text.align_right)
    table.cell(T, 0, 3, "Take profit", text_color = color.gray, text_halign = text.align_left)
    table.cell(T, 1, 3, fp(tp) + "  (+$" + str.tostring(win, "#.0") + ")", text_color = color.green, text_halign = text.align_right)
    table.cell(T, 0, 4, "Odds (tested)", text_color = color.gray, text_halign = text.align_left)
    table.cell(T, 1, 4, "~1 in 4 wins, ~10–15 min", text_color = color.silver, text_halign = text.align_right)
    table.cell(T, 0, 5, "After a win", text_color = color.gray, text_halign = text.align_left)
    table.cell(T, 1, 5, "Withdraw right away", text_color = color.silver, text_halign = text.align_right)
