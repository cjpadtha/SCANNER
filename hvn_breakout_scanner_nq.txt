//  • Not big → no trade, just a printed buy/sell probability

    float pl = ta.pivotlow(low, len, len)

    if not na(ph)

        trend := 1
    if not na(lastL) and close < lastL

    trend

// ───────────────────────── HVN zones ─────────────────────────
// Bins each bar's volume by its close over `n` bars starting `off` bars back.
// Adjacent high-volume rows merge into one zone; zone top/bottom = highest/lowest close in it.
f_zones(int off, int n, int nRows, float mult) =>
    float hi = high[off]
    float lo = low[off]
    for i = off to off + n - 1
        hi := math.max(hi, high[i])
        lo := math.min(lo, low[i])
    float step = (hi - lo) / nRows
    vols = array.new_float(nRows, 0.0)
    cMax = array.new_float(nRows, na)
    cMin = array.new_float(nRows, na)
    tops = array.new_float()
    bots = array.new_float()
    zv   = array.new_float()
    if step > 0
        for i = off to off + n - 1
            int idx = math.max(0, math.min(nRows - 1, math.floor((close[i] - lo) / step)))
            array.set(vols, idx, array.get(vols, idx) + nz(volume[i]))
            float cx = array.get(cMax, idx)
            array.set(cMax, idx, na(cx) ? close[i] : math.max(cx, close[i]))
            float cn = array.get(cMin, idx)
            array.set(cMin, idx, na(cn) ? close[i] : math.min(cn, close[i]))
        float thr = array.avg(vols) * mult
        float gTop = na
        float gBot = na
        float gVol = 0.0
        for r = 0 to nRows - 1
            float v = array.get(vols, r)
            bool isH = v > 0 and v >= thr
            if isH
                gTop := na(gTop) ? array.get(cMax, r) : math.max(gTop, array.get(cMax, r))
                gBot := na(gBot) ? array.get(cMin, r) : math.min(gBot, array.get(cMin, r))
                gVol += v
            if (not isH or r == nRows - 1) and not na(gTop)
                array.push(tops, gTop)
                array.push(bots, gBot)
                array.push(zv, gVol)
                gTop := na
                gBot := na
                gVol := 0.0
    [tops, bots, zv]

f_add(array<float> a, int i, float v) =>
    array.set(a, i, array.get(a, i) + v)
f_bit(int m, int i) =>
    math.floor(m / math.pow(2, i)) % 2 == 1
f_bucket(float sc) =>
    sc < 50 ? 0 : sc < 65 ? 1 : sc < 80 ? 2 : 3
f_r(float v) =>
    (v >= 0 ? "+" : "") + str.tostring(v, "#.##") + "R"

if array.size(pDir) > 0
    for k = array.size(pDir) - 1 to 0
        if bar_index > array.get(pBar, k)
            int   sd = array.get(pDir, k)
            float se = array.get(pEntry, k)
            float ss = array.get(pStop, k)
            float sTg = array.get(pTgt, k)
            float rk = math.abs(se - ss)
            bool hitSl   = sd == 1 ? low <= ss : high >= ss
            bool hitTp   = sd == 1 ? high >= sTg : low <= sTg
            bool expired = bar_index - array.get(pBar, k) >= maxHold

                aN += 1.0
                aW += win
                aR += net
                int bk = array.get(pBucket, k)

                f_add(bW, bk, win)
                f_add(bR, bk, net)
                int si = array.get(pSess, k)

                f_add(sW, si, win)
                f_add(sR, si, net)

                    if f_bit(fl, i)
                        f_add(fyN, i, 1.0)
                        f_add(fyW, i, win)
                        f_add(fyR, i, net)
                    else
                        f_add(fnN, i, 1.0)
                        f_add(fnW, i, win)
                        f_add(fnR, i, net)
                array.remove(pDir, k)
                array.remove(pEntry, k)
                array.remove(pStop, k)
                array.remove(pTgt, k)
                array.remove(pBar, k)
